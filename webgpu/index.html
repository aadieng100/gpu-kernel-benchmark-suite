// WGSL Compute Shader: Workgroup Shared Memory Tiled GEMM
const TILE_SIZE : u32 = 16u;
var<workgroup> As : array<array<f32, 16>, 16>;
var<workgroup> Bs : array<array<f32, 16>, 16>;

@compute @workgroup_size(16, 16)
fn main(...) {
    // Collaborative load into Workgroup Shared Memory
    As[ty][tx] = A[row * K + t * TILE_SIZE + tx];
    Bs[ty][tx] = B[(t * TILE_SIZE + ty) * N + col];

    workgroupBarrier(); // Synchronize workgroup SRAM

    for (var k : u32 = 0u; k < TILE_SIZE; k = k + 1u) {
        acc = acc + As[ty][k] * Bs[k][tx];
    }
    workgroupBarrier();
}
// WGSL Compute Shader: Naive GEMM
@compute @workgroup_size(16, 16)
fn main(@builtin(global_invocation_id) global_id : vec3<u32>) {
    let col = global_id.x;
    let row = global_id.y;

    if (row < dims.M && col < dims.N) {
        var sum : f32 = 0.0;
        for (var k : u32 = 0u; k < dims.K; k = k + 1u) {
            sum = sum + A[row * dims.K + k] * B[k * dims.N + col];
        }
        C[row * dims.N + col] = sum;
    }
}
